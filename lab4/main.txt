#include <iostream>
#include <chrono>
#include "header.h"

using namespace std;

int main()
{
    int n=8;
    int a[100]={1,2,3,4,5,6,7,8};

    cout<<"crescator: ";
    auto start=chrono::high_resolution_clock::now();
    
    batcher(a,0,n,1);
    
    auto end=chrono::high_resolution_clock::now();
    chrono::duration<double,std::milli> diff=end-start;
    
    cout<<endl;

    for(int i=0;i<n;i++)
        cout<<a[i]<<" ";
    
    cout<<endl;
    cout<<"timp: "<<diff.count()<<"ms";
    cout<<endl<<endl;

    int b[]={6,5,4,3,2,1,10,7};
    cout<<"descrescator:";
    cout<<endl;
    
    start=chrono::high_resolution_clock::now();
    
    batcher(b,0,n,0);
    
    end=chrono::high_resolution_clock::now();
    
    diff=end-start;
    for(int i=0;i<n;i++)
        cout<<b[i]<<" ";
    cout<<endl;
    cout<<"timp: "<<diff.count()<<" ms";
    
    cout<<endl<<endl;

    int c[]={2,9,5,7,1,4,3,8};
    cout<<"aleator:";
    cout<<endl;
    
    start=chrono::high_resolution_clock::now();
    
    batcher(c,0,n,1);
    
    end=chrono::high_resolution_clock::now();
    
    diff=end-start;
    for(int i=0;i<n;i++)
        cout<<c[i]<<" ";
    cout<<endl;
    cout<<"timp: "<<diff.count()<<" ms";

    return 0;
}
