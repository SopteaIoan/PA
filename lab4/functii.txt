#include <iostream>

using namespace std;

void compara_si_schimba(int &x,int &y,int s)
{
    if ((s==1 && x>y) || (s==0 && x<y)) 
    {
        int z=x;
        x=y;
        y=z;
    }
}

void sortarebitona(int a[],int i,int d,int s)
{
    if (d==2)
    {
        compara_si_schimba(a[i],a[i+1],s);
    }
    else
    {
        for (int j=0;j<=d/2-1;j++)
        {
            compara_si_schimba(a[i+j],a[i+j+d/2],s);
        }
        sortarebitona(a,i,d/2,s);
        sortarebitona(a,i+d/2,d/2,s);
    }
}

void batcher(int a[],int i,int d,int s)
{
    if (d==2)
    {
        compara_si_schimba(a[i],a[i+1],s);
    }
    else if (d>2)
    {
        batcher(a,i,d/2,1);
        batcher(a,i+d/2,d/2,0);
        
        sortarebitona(a,i,d,s);
    }
}
