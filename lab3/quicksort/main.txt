#include <iostream>
#include "header.h"
using namespace std;

int main() 
{
    int v[]={10,36,5,99,4,25,2};
    int n=7;

    cout<<"vectorul initial: ";
    for (int i = 0; i < n; i++)
    {
        cout<<v[i]<<" ";
    }
    cout<<endl;
    
    quicksort(v,0,n-1);

    cout<<"vectorul sortat: ";
    for (int i=0;i<n;i++)
    {
        cout<<v[i]<<" ";
    }
    return 0;
}
