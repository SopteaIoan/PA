#include <iostream>
using namespace std;

void interclasare(int v[],int p,int q,int m,int temp[])
{
    int i=p;
    int j=m+1;
    int k=-1;

    while ((i<=m)&&(j<=q)) 
    {
        k=k+1;
        if (v[i]<=v[j])
        {
            temp[k]=v[i];
            i=i+1;
        } 
        else
        {
            temp[k]=v[j];
            j=j+1;
        }
    }

    while (i<=m)
    {
        k=k+1;
        temp[k]=v[i];
        i=i+1;
    }

    while (j<=q)
    {
        k=k+1;
        temp[k]=v[j];
        j=j+1;
    }
}

void mergesort(int v[],int p,int q,int temp[])
{
    if (p<q) {
        int m=(p+q)/2; 

        mergesort(v,p,m,temp);
        mergesort(v,m+1,q,temp);
        
        interclasare(v,p,q,m,temp);

        for (int i=p;i<=q;i++)
        {
            v[i]=temp[i-p];
        }
    }
}
